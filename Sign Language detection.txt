# LANDMARK FEATURE EXTRACTION

def extract_hand_features(frame):

    rgb = cv2.cvtColor(
        frame,
        cv2.COLOR_BGR2RGB
    )

    results = hands_detector.process(
        rgb
    )

    if not results.multi_hand_landmarks:

        return None, results

    all_features = []

    # Maximum two hands

    for hand_landmarks in results.multi_hand_landmarks:

        landmarks = []

        for landmark in hand_landmarks.landmark:

            landmarks.append(
                landmark.x
            )

            landmarks.append(
                landmark.y
            )

            landmarks.append(
                landmark.z
            )

        all_features.extend(
            landmarks
        )

    # Make fixed length feature vector

    expected_length = 126

    while len(all_features) < expected_length:

        all_features.append(0.0)

    all_features = all_features[
        :expected_length
    ]

    return (
        np.array(
            all_features,
            dtype=np.float32
        ),
        results
    )

                [

                    for i in range(
                        len(features)
                    )

        writer.writerow(
