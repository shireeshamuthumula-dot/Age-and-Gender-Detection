def detect_dress_colour(image):
    """
    Estimate the dominant colour in the lower part
    of the image.

    This is a simple image-processing method.
    """

    height, width = image.shape[:2]

    # Use lower 55% of image
    lower_part = image[
        int(height * 0.45):height,
        :
    ]

    if lower_part.size == 0:
        return "Unknown"

    # Resize
    lower_part = cv2.resize(
        lower_part,
        (100, 100)
    )

    hsv = cv2.cvtColor(
        lower_part,
        cv2.COLOR_BGR2HSV
    )

    # Remove very dark pixels
    valid = hsv[:, :, 2] > 40

    if np.count_nonzero(valid) < 100:
        return "Black"

    hue = hsv[:, :, 0]

    saturation = hsv[:, :, 1]

    value = hsv[:, :, 2]

    
    # Colour masks

    red1 = cv2.inRange(
        hsv,
        np.array([0, 70, 40]),
        np.array([10, 255, 255])
    )

    red2 = cv2.inRange(
        hsv,
        np.array([170, 70, 40]),
        np.array([180, 255, 255])
    )

    red = red1 | red2

    blue = cv2.inRange(
        hsv,
        np.array([90, 60, 40]),
        np.array([140, 255, 255])
    )

    green = cv2.inRange(
        hsv,
        np.array([35, 50, 40]),
        np.array([85, 255, 255])
    )

    yellow = cv2.inRange(
        hsv,
        np.array([15, 60, 40]),
        np.array([35, 255, 255])
    )

    purple = cv2.inRange(
        hsv,
        np.array([140, 40, 40]),
        np.array([170, 255, 255])
    )

    total = np.count_nonzero(valid)

    percentages = {
        "Red": np.count_nonzero(
            red & valid
        ) / total,

        "Blue": np.count_nonzero(
            blue & valid
        ) / total,

        "Green": np.count_nonzero(
            green & valid
        ) / total,

        "Yellow": np.count_nonzero(
            yellow & valid
        ) / total,

        "Purple": np.count_nonzero(
            purple & valid
        ) / total
    }

    best_colour = max(
        percentages,
        key=percentages.get
    )

    if percentages[best_colour] > 0.12:
        return best_colour

    mean_value = np.mean(
        value[valid]
    )

    mean_saturation = np.mean(
        saturation[valid]
    )

    if mean_value < 65:
        return "Black"

    if (
        mean_value > 185
        and
        mean_saturation < 65
    ):
        return "White"

    if mean_saturation < 55:
        return "Gray"

    return "Other"

# CONDITIONAL LOGIC

def analyze_image(
    image,
    category
):
    """
    Apply project rules according to the
    user-selected category.
    """

    result = {}

        # Always predict emotion
    
    result["Emotion"] = detect_emotion(
        image
    )

    
    # Category 1
    
    if category == "Indian":

        result["Nationality"] = (
            "Indian (user selected)"
        )

        result["Age"] = (
            estimate_age_group(
                image
            )
        )

        result["Dress Colour"] = (
            detect_dress_colour(
                image
            )
        )

    
    # Category 2
    
    elif category == "United States":

        result["Nationality"] = (
            "United States (user selected)"
        )

        result["Age"] = (
            estimate_age_group(
                image
            )
        )

        # Category 3
    
    elif category == "African":

        result["Nationality"] = (
            "African (user selected)"
        )

        result["Dress Colour"] = (
            detect_dress_colour(
                image
            )
        )

        # Other
    

    else:

        result["Nationality"] = (
            category
            + " (user selected)"
        )

    return result

    def upload_image(self):

        filename = filedialog.askopenfilename(
            title="Select Image",
            filetypes=[
                (
                    "Image Files",
                    "*.jpg *.jpeg *.png *.bmp"
                ),
                (
                    "All Files",
                    "*.*"
                )
            ]
        )

        if not filename:
            return

        image = cv2.imread(
            filename
        )

        if image is None:

            messagebox.showerror(
                "Error",
                "Could not read the image."
            )

            return

        self.image = image

        self.show_image(
            image
        )

        self.output.config(
            text=(
                "Image uploaded successfully.\n\n"
                "Select a category and click\n"
                "ANALYZE IMAGE."
            )
        )

        # SHOW IMAGE
    
    def show_image(
        self,
        image
    ):

        rgb = cv2.cvtColor(
            image,
            cv2.COLOR_BGR2RGB
        )

        pil_image = Image.fromarray(
            rgb
        )

        pil_image.thumbnail(
            (500, 400)
        )

        self.photo = ImageTk.PhotoImage(
            pil_image
        )

        self.image_label.config(
            image=self.photo,
            text=""
        )

        # ANALYZE
    
    def analyze(self):

        if self.image is None:

            messagebox.showwarning(
                "No Image",
                "Please upload an image first."
            )

            return

        category = (
            self.category.get()
        )

        results = analyze_image(
            self.image,
            category
        )

        output_text = (
            "ANALYSIS RESULT\n"
            "========================\n\n"
        )

        for key, value in results.items():

            output_text += (
                f"{key}: {value}\n\n"
            )

        self.output.config(
            text=output_text
        )

        # CLEAR
    
    def clear(self):

        self.image = None

        self.photo = None

        self.image_label.config(
            image="",
            text="No image selected"
        )

        self.output.config(
            text=(
                "Upload an image\n"
                "and click ANALYZE IMAGE."
            )
        )
